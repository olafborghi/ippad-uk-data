Explore

How different groups of young people answered the questions of the four surveys.

This page shows how the young people in the four surveys answered most of our questions, and how their answers differ between groups. You can choose a question and a characteristic to compare by, such as age or gender, and limit the data to certain surveys or ages. All calculations run in your browser, on the same data that the data page describes.

Because the surveys differ in age range, recruitment and timing, the explorer shows each survey in its own panel unless you choose to pool them. In pooled data, a difference between two groups may partly reflect who took part in each survey rather than a real difference between the groups. The address of the page changes with your choices, so you can share a link to the chart you are looking at.

The samples are not weighted, so the results describe the young people who took part rather than all young people in the UK. The four surveys also differ in their design, and differences between them should therefore be read with care. The data page explains how each survey was carried out and what to keep in mind when comparing them.

respondentsRaw = FileAttachment("data/respondents.csv").csv()
variablesRaw = FileAttachment("data/variables.csv").csv()
levelsRaw = FileAttachment("data/levels.csv").csv()
compareByRaw = FileAttachment("data/compare_by.csv").csv()

// categories of every factor, in their order
levelsOf = d3.rollup(
  levelsRaw.slice().sort((a, b) => +a.order - +b.order),
  v => v.map(d => d.level),
  d => d.variable
)

datasets = levelsOf.get("dataset")

// "Gallup W1" becomes "gallup-w1", the name of its colour in _brand.yml
datasetKey = ds => ds.toLowerCase().replace(/\s+/g, "-")

datasetColour = new Map(datasets.map(ds => [ds, brand[datasetKey(ds)]]))

variables = new Map(variablesRaw.map(d => [d.name, {
  ...d,
  min: d.min === "" ? null : +d.min,
  max: d.max === "" ? null : +d.max,
  in_explorer: d.in_explorer === "TRUE",
  nBySurvey: new Map(datasets.map(ds => [ds, +d["n_" + datasetKey(ds).replace("-", "_")]]))
}]))

// one row per interview, with numbers, TRUE/FALSE and empty fields converted
respondents = respondentsRaw.map(d => {
  const row = {};
  for (const key in d) {
    const value = d[key];
    const type = variables.get(key)?.type;
    row[key] = value === "" ? null
      : type === "number" ? +value
      : type === "logical" ? value === "TRUE"
      : value;
  }
  return row;
})

ageExtent = d3.extent(respondents, d => d.age)
nTwice = respondents.filter(d => !d.person_first).length

// explorer variables grouped by topic, topics in the order of the questionnaire
topics = d3.groups(
  variablesRaw.map((d, i) => ({...d, i})).filter(d => d.in_explorer === "TRUE"),
  d => d.topic
).sort((a, b) => d3.mean(a[1], d => d.i) - d3.mean(b[1], d => d.i))

// Settings from the address of the page ---------------------------------

init = {
  const p = new URLSearchParams(window.location.search);
  const v = p.get("var");
  const by = p.get("by");
  const keys = p.get("surveys")?.split(",") ?? [];

    age: ageOk ? age.sort((a, b) => a - b) : ageExtent,

// The settings in use ---------------------------------------------------

current = variables.get(varName)
varLevels = levelsOf.get(varName) ?? []
dkLevel = varLevels.find(l => /don.t know/i.test(l))
scaleLevels = varLevels.filter(l => l !== dkLevel)
kind = current.type === "number" ? "number"
  : current.type === "logical" ? "logical"
  : current.type === "ordered" || dkLevel ? "diverging"
  : "categorical"
shownLevels = dkLevel && showDk ? varLevels : scaleLevels
byLabel = compareByRaw.find(d => d.name === by).label

surveyComparison = ["dataset", "provider", "wave"].includes(by)
splitActive = split && !surveyComparison

askedIn = surveys.filter(ds => current.nBySurvey.get(ds) > 0)
notAsked = surveys.filter(ds => current.nBySurvey.get(ds) === 0)

groupOf = by === "wave" ? (d => d.wave == null ? null : `Wave ${d.wave}`)
  : by === "age" ? (d => d.age == null ? null : String(d.age))
  : (d => d[by])
groupDomain = by === "age" ? d3.range(ageRange[0], ageRange[1] + 1).map(String)
  : by === "wave" ? ["Wave 1", "Wave 2"]
  : levelsOf.get(by)

selection = respondents.filter(d =>
  askedIn.includes(d.dataset) &&
  d.age >= ageRange[0] && d.age <= ageRange[1] &&
  (!once || d.person_first)
)
answered = selection.filter(d => d[varName] != null)
nNoGroup = answered.filter(d => groupOf(d) == null).length
nDkLeftOut = dkLevel && !showDk ? answered.filter(d => d[varName] === dkLevel).length : 0
answers = answered.filter(d =>
  groupOf(d) != null && (!dkLevel || showDk || d[varName] !== dkLevel)
)

minN = 20

// Summaries per group ---------------------------------------------------

scale = {
  if (kind !== "number") return null;
  const min = current.min ?? d3.min(answers, d => d[varName]);
  const max = current.max ?? d3.max(answers, d => d[varName]);
  const binWidth = max - min > 20 ? 5 : 1;
  const nBins = binWidth === 1 ? max - min + 1 : Math.ceil((max - min) / binWidth);
  return {
    min, max, binWidth, nBins,
    bin: v => Math.min(Math.floor((v - min) / binWidth), nBins - 1),
    centre: k => binWidth === 1 ? min + k : min + (k + 0.5) * binWidth
  };
}

summarise = (rows, colour) => {
  const n = rows.length;
  const base = {n, colour, faded: n < minN};
  if (kind === "number") {
    const values = rows.map(d => d[varName]);
    const mean = d3.mean(values);
    const sd = d3.deviation(values);
    const se = n > 1 ? sd / Math.sqrt(n) : NaN;
    const counts = new Array(scale.nBins).fill(0);
    for (const v of values) counts[scale.bin(v)] += 1;
    return {...base, mean, sd, lo: mean - 1.96 * se, hi: mean + 1.96 * se,
      dist: counts.map((c, k) => ({k, share: c / n}))};
  }
  if (kind === "logical") {
    const yes = rows.filter(d => d[varName]).length;
    const p = yes / n;
    const se = Math.sqrt(p * (1 - p) / n);
    return {...base, yes, p, lo: Math.max(0, p - 1.96 * se), hi: Math.min(1, p + 1.96 * se)};
  }
  const counts = d3.rollup(rows, v => v.length, d => d[varName]);
  const shares = shownLevels.map(level => ({
    level, count: counts.get(level) ?? 0, share: (counts.get(level) ?? 0) / n
  }));
  // position of the diverging bar: the lower half left of zero, the middle answer split across it
  const k = scaleLevels.length;
  const m = Math.floor(k / 2);
  const s = scaleLevels.map(l => (counts.get(l) ?? 0) / n);
  const left = d3.sum(s.slice(0, m));
  const right = d3.sum(s.slice(k - m));
  const start = -(left + (k % 2 ? s[m] / 2 : 0));
  return {...base, shares, left, right, start, end: start + d3.sum(s)};
}

panels = {
  const specs = splitActive
    ? askedIn.map(ds => ({key: ds, title: ds, colour: datasetColour.get(ds), rows: answers.filter(d => d.dataset === ds)}))
    : [{key: "all", title: null, colour: brand["blue-dark"], rows: answers}];
  return specs.map(p => {
    const byGroup = d3.group(p.rows, groupOf);
    const groups = groupDomain
      .filter(g => byGroup.has(g))
      .map((g, i) => ({
        group: g,
        i,
        survey: p.title,
        ...summarise(byGroup.get(g), by === "dataset" ? datasetColour.get(g) : p.colour)
      }));
    return {...p, n: p.rows.length, groups};
  });
}

shownPanels = panels.filter(p => p.groups.length > 0)
emptySurveys = splitActive ? panels.filter(p => p.groups.length === 0).map(p => p.key) : []
allGroups = shownPanels.flatMap(p => p.groups)

// Formatting -------------------------------------------------------------

fmtN = d3.format(",")
fmtPct = v => `${Math.round(v * 100)}%`
fmtMean = d3.format(".1f")
joinWords = xs => xs.length <= 1 ? (xs[0] ?? "") : `${xs.slice(0, -1).join(", ")} and ${xs[xs.length - 1]}`

anchors = (current.anchors || "")
  .split(/,\s*(?=-?\d+\s*=)/)
  .map(s => s.match(/^\s*(-?\d+)\s*=\s*(.+?)\s*$/))
  .filter(Boolean)
  .map(m => ({value: +m[1], text: m[2]}))

anchorSentence = {
  if (anchors.length < 2) return null;
  const first = anchors[0];
  const last = anchors[anchors.length - 1];
  const middle = anchors.slice(1, -1).map(a => `${a.value} means “${a.text}”`);
  return `The scale runs from ${first.value} (“${first.text}”) to ${last.value} (“${last.text}”)` +
    (middle.length ? `, and ${joinWords(middle)}.` : ".");
}

// diverging colours: coral for the lower half, grey in the middle, blue for the upper half
levelColour = {
  const k = scaleLevels.length;
  const m = Math.floor(k / 2);
  const grey = d3.interpolateLab(brand.rule, brand.muted)(0.2);
  const coralLight = d3.interpolateLab(brand.coral, brand.paper)(0.5);
  const blueLight = d3.interpolateLab(brand.blue, brand.paper)(0.45);
  const ramp = (dark, base, light) =>
    m === 1 ? [base] : m === 2 ? [dark, base] : m === 3 ? [dark, base, light]
    : d3.quantize(d3.interpolateLab(dark, light), m);
  const left = ramp(brand["coral-dark"], brand.coral, coralLight);
  const right = ramp(brand["blue-dark"], brand.blue, blueLight).reverse();
  const colours = [...left, ...(k % 2 ? [grey] : []), ...right];
  const map = new Map(scaleLevels.map((l, j) => [l, colours[j]]));
  if (kind === "categorical") {
    const cats = [brand["blue-dark"], brand.coral, brand.green, brand.yellow, brand.blue, brand["coral-dark"], brand.muted];
    scaleLevels.forEach((l, j) => map.set(l, cats[j % cats.length]));
  }
  if (dkLevel) map.set(dkLevel, d3.interpolateLab(brand.muted, brand.paper)(0.3));
  return map;
}

drawNumber = (p, isLast) => {
  const G = p.groups;
  const rowH = 52;
  const {min, max, binWidth} = scale;
  const span = binWidth === 1 ? [min - 0.5, max + 0.5] : [min, max];
  const ticks = max - min <= 10 ? d3.range(min, max + 1) : d3.ticks(min, max, chartWidth < 560 ? 5 : 10);
  const histMax = d3.max(allGroups, g => d3.max(g.dist, b => b.share));
  const base = 0.76;
  const bars = G.flatMap(g => g.dist.filter(b => b.share > 0).map(b => ({...b, g})));
  const showAnchors = isLast && anchors.length > 0;
  const marginBottom = isLast ? (showAnchors ? 74 : 46) : 24;
  const binText = k => binWidth === 1 ? `${scale.centre(k)}` : `${min + k * binWidth} to ${Math.min(max, min + (k + 1) * binWidth - (k === scale.nBins - 1 ? 0 : 1))}`;
  return Plot.plot({
    width: chartWidth,
    height: G.length * rowH + 6 + marginBottom,
    marginTop: 6, marginBottom, marginLeft: layout.marginLeft, marginRight: layout.marginRight,
    style: layout.style,
    x: {domain: span, ticks, tickFormat: d3.format("~g"), tickSize: 0, tickPadding: 6,
      label: isLast ? `Average answer, on the scale from ${min} to ${max}` : null,
      labelAnchor: "center", labelArrow: "none", labelOffset: showAnchors ? 68 : 40},
    y: {domain: [0, G.length], reverse: true, axis: null},
    marks: [
      Plot.gridX(ticks, {stroke: brand.rule, strokeOpacity: 1}),
      Plot.rect(bars, {
        x1: d => scale.centre(d.k) - 0.4 * binWidth, x2: d => scale.centre(d.k) + 0.4 * binWidth,

drawDiverging = (p, isLast, width, withN) => {
  const G = p.groups;
  const rowH = 36;
  const k = scaleLevels.length;
  const m = Math.floor(k / 2);
  const segments = G.flatMap(g => {
    let x = g.start;
    return scaleLevels.map(level => {
      const s = g.shares.find(d => d.level === level);
      const seg = {g, level, share: s.share, count: s.count, x1: x, x2: x + s.share};
      x += s.share;
      return seg;
    });
  });
  const reach = d3.max(allGroups, g => Math.max(-g.start, g.end));
  const D = Math.min(1, Math.ceil(reach * 10 - 1e-9) / 10);
  const marginRight = withN ? layout.marginRight : 10;
  const plotW = width - layout.marginLeft - marginRight;
  const pad = 2 * D * 36 / Math.max(80, plotW - 72);
  const ticks = d3.ticks(-D, D, plotW < 300 ? 4 : 8);
  return Plot.plot({
    width,
    height: G.length * rowH + 6 + (isLast ? 44 : 22),
    marginTop: 6, marginBottom: isLast ? 44 : 22, marginLeft: layout.marginLeft, marginRight,
    style: layout.style,
    x: {domain: [-D - pad, D + pad], ticks, tickFormat: v => fmtPct(Math.abs(v)), tickSize: 0, tickPadding: 6,
      label: isLast ? "Share of answers" : null, labelAnchor: "center", labelArrow: "none", labelOffset: 38},
    y: {domain: [0, G.length], reverse: true, axis: null},
    marks: [
      Plot.gridX(ticks, {stroke: brand.rule, strokeOpacity: 1}),
      Plot.rect(segments, {
        x1: "x1", x2: "x2", y1: d => d.g.i + 0.18, y2: d => d.g.i + 0.82,
        fill: d => levelColour.get(d.level), fillOpacity: d => d.g.faded ? 0.4 : 1,
        stroke: brand.paper, strokeWidth: 0.75,
        title: d => `${d.g.group}, ${d.level}: ${fmtPct(d.share)} (${fmtN(d.count)} of ${fmtN(d.g.n)})`
      }),
      Plot.ruleX([0], {stroke: brand.ink, strokeWidth: 1}),
      m > 0 ? Plot.text(G, {x: "start", y: d => d.i + 0.5, text: d => fmtPct(d.left), textAnchor: "end", dx: -5,
        fill: brand["coral-dark"], fillOpacity: d => d.faded ? 0.5 : 1, fontWeight: 600}) : null,
      m > 0 ? Plot.text(G, {x: "end", y: d => d.i + 0.5, text: d => fmtPct(d.right), textAnchor: "start", dx: 5,
        fill: brand["blue-dark"], fillOpacity: d => d.faded ? 0.5 : 1, fontWeight: 600}) : null,
      ...rowLabels(G, withN)
    ]
  });
}

// "I don't know" as a narrow chart of its own, next to the diverging bars
drawDk = (p, isLast, width) => {
  const G = p.groups.map(g => ({...g, dk: g.shares.find(s => s.level === dkLevel)}));
  const rowH = 36;
  const dkMax = d3.max(allGroups, g => g.shares.find(s => s.level === dkLevel).share);
  const D = Math.max(0.1, Math.ceil(dkMax * 10 - 1e-9) / 10);
  const plotW = width - 8 - layout.marginRight;
  return Plot.plot({
    width,
    height: G.length * rowH + 6 + (isLast ? 44 : 22),
    marginTop: 6, marginBottom: isLast ? 44 : 22, marginLeft: 8, marginRight: layout.marginRight,
    style: layout.style,
    x: {domain: [0, D * (1 + 34 / Math.max(30, plotW - 34))], ticks: [], tickSize: 0,
      label: isLast ? "Don't know" : null, labelAnchor: "center", labelArrow: "none", labelOffset: 38},
    y: {domain: [0, G.length], reverse: true, axis: null},
    marks: [
      Plot.gridX([0, D], {stroke: brand.rule, strokeOpacity: 1}),
      Plot.rect(G, {
        x1: 0, x2: d => d.dk.share, y1: d => d.i + 0.18, y2: d => d.i + 0.82,
        fill: levelColour.get(dkLevel), fillOpacity: d => d.faded ? 0.4 : 1,
        title: d => `${d.group}, ${dkLevel}: ${fmtPct(d.dk.share)} (${fmtN(d.dk.count)} of ${fmtN(d.n)})`
      }),
      Plot.text(G, {x: d => d.dk.share, y: d => d.i + 0.5, text: d => fmtPct(d.dk.share), textAnchor: "start", dx: 5,
        fill: brand.muted, fillOpacity: d => d.faded ? 0.5 : 1, fontWeight: 600}),
      layout.showN ? Plot.text(G, {frameAnchor: "right", y: d => d.i + 0.5, text: d => `n = ${fmtN(d.n)}`, textAnchor: "start", dx: 10,
        fill: brand.muted, fontSize: 11}) : null
    ]
  });
}

drawLogical = (p, isLast) => {
  const G = p.groups;
  const rowH = 34;
  const top = d3.max(allGroups, g => g.hi);
  const D = Math.min(1, Math.max(0.1, Math.ceil(top * 10 - 1e-9) / 10));
  const plotW = chartWidth - layout.marginLeft - layout.marginRight;
  const ticks = d3.ticks(0, D, plotW < 300 ? 4 : 6);
  return Plot.plot({
    width: chartWidth,
    height: G.length * rowH + 6 + (isLast ? 44 : 22),
    marginTop: 6, marginBottom: isLast ? 44 : 22, marginLeft: layout.marginLeft, marginRight: layout.marginRight,
    style: layout.style,
    x: {domain: [0, D * (1 + 40 / Math.max(60, plotW - 40))], ticks, tickFormat: fmtPct, tickSize: 0, tickPadding: 6,
      label: isLast ? "Share who ticked this option, with 95% confidence interval" : null,
      labelAnchor: "center", labelArrow: "none", labelOffset: 38},
    y: {domain: [0, G.length], reverse: true, axis: null},
    marks: [
      Plot.gridX(ticks, {stroke: brand.rule, strokeOpacity: 1}),
      Plot.rect(G, {
        x1: 0, x2: "p", y1: d => d.i + 0.22, y2: d => d.i + 0.78, fill: "colour", fillOpacity: d => d.faded ? 0.35 : 0.9,
        title: d => `${d.group}: ${fmtPct(d.p)} (${fmtN(d.yes)} of ${fmtN(d.n)})`
      }),
      Plot.link(G, {x1: "lo", x2: "hi", y1: d => d.i + 0.5, y2: d => d.i + 0.5, stroke: brand.ink, strokeWidth: 1.2,
        strokeOpacity: d => d.faded ? 0.4 : 0.8}),
      Plot.text(G, {x: "hi", y: d => d.i + 0.5, text: d => fmtPct(d.p), textAnchor: "start", dx: 6,
        fill: brand.ink, fillOpacity: d => d.faded ? 0.5 : 1, fontWeight: 600}),
      Plot.ruleX([0], {stroke: brand.ink}),
      ...rowLabels(G)
    ]
  });
}

drawCategorical = (p, isLast) => {
  const G = p.groups;
  const rowH = 36;
  const segments = G.flatMap(g => {
    let x = 0;
    return g.shares.map(s => ({g, ...s, x1: x, x2: (x += s.share)}));
  });
  return Plot.plot({
    width: chartWidth,
    height: G.length * rowH + 6 + (isLast ? 44 : 22),
    marginTop: 6, marginBottom: isLast ? 44 : 22, marginLeft: layout.marginLeft, marginRight: layout.marginRight,
    style: layout.style,
    x: {domain: [0, 1], tickFormat: fmtPct, tickSize: 0, tickPadding: 6, ticks: 5,
      label: isLast ? "Share of answers" : null, labelAnchor: "center", labelArrow: "none", labelOffset: 38},
    y: {domain: [0, G.length], reverse: true, axis: null},
    marks: [
      Plot.rect(segments, {
        x1: "x1", x2: "x2", y1: d => d.g.i + 0.18, y2: d => d.g.i + 0.82,
        fill: d => levelColour.get(d.level), fillOpacity: d => d.g.faded ? 0.4 : 1, stroke: brand.paper, strokeWidth: 0.75,
        title: d => `${d.g.group}, ${d.level}: ${fmtPct(d.share)}`
      }),
      ...rowLabels(G)
    ]
  });
}

makeLegend = () => {
  if (kind !== "diverging" && kind !== "categorical") return null;
  const items = [...scaleLevels, ...(dkLevel && showDk ? [dkLevel] : [])];
  return htl.html`<div class="ex-legend" aria-label="Legend">${items.map(l =>
    htl.html`<span><span class="ex-swatch" style=${{background: levelColour.get(l)}}></span>${l}</span>`)}</div>`;
}

drawPanel = (p, isLast) => {
  if (kind === "number") return drawNumber(p, isLast);
  if (kind === "logical") return drawLogical(p, isLast);
  if (kind === "categorical") return drawCategorical(p, isLast);
  if (dkLevel && showDk) {
    const dkWidth = Math.round(layout.showN ? Math.max(116, chartWidth * 0.19) : 64);
    return htl.html`<div class="ex-panel">${drawDiverging(p, isLast, chartWidth - dkWidth, false)}${drawDk(p, isLast, dkWidth)}</div>`;
  }
  return drawDiverging(p, isLast, chartWidth, true);
}

// What the visitor sees --------------------------------------------------

problem = {
  if (!surveys.length) return "Please select at least one survey.";
  if (!askedIn.length) {
    const asked = datasets.filter(ds => current.nBySurvey.get(ds) > 0);
    return `${joinWords(notAsked)} did not ask this question. It was asked in ${joinWords(asked)}, and you can include these surveys in the settings.`;
  }
  if (by === varName) return "This question is also the comparison, so the chart would show nothing new. Please choose another characteristic to compare by.";
  if (!answers.length) return "No respondents match the current selection. You could widen the age range or include more surveys.";
  return null;
}

howToRead = {
  if (kind === "number") return "Each dot shows the average answer of a group, and the coloured line around it the 95% confidence interval. The grey bars above the line show how the answers of the group are spread over the scale.";
  if (kind === "logical") return "Each bar shows the share of respondents in a group who ticked this option, and the thin line its 95% confidence interval.";
  if (kind === "categorical") return "Each bar shows how the answers of a group are divided between the categories.";
  const k = scaleLevels.length;
  const m = Math.floor(k / 2);
  const sides = `Answers from “${scaleLevels[0]}” to “${scaleLevels[m - 1]}” extend to the left of the centre line, and answers from “${scaleLevels[k - m]}” to “${scaleLevels[k - 1]}” to the right.`;
  const middle = k % 2 ? ` The middle answer, “${scaleLevels[m]}”, is split evenly across the line.` : "";
  const dk = dkLevel && showDk ? ` The share who answered “${dkLevel}” is shown separately on the right.` : "";
  return `Each bar shows how the answers of a group are spread over the scale. ${sides}${middle} The numbers at either end give the total share on each side.${dk}`;
}

makeTable = () => {
  const rows = shownPanels.flatMap(p => p.groups);
  let head, cells;
  if (kind === "number") {
    head = ["Answers", "Average", "95% CI from", "to", "Standard deviation"];
    cells = d => [fmtN(d.n), d3.format(".2f")(d.mean), d3.format(".2f")(d.lo), d3.format(".2f")(d.hi), d3.format(".2f")(d.sd)];
  } else if (kind === "logical") {
    head = ["Answers", "Ticked", "Share", "95% CI from", "to"];
    cells = d => [fmtN(d.n), fmtN(d.yes), d3.format(".1%")(d.p), d3.format(".1%")(d.lo), d3.format(".1%")(d.hi)];
  } else {
    head = ["Answers", ...shownLevels];
    cells = d => [fmtN(d.n), ...d.shares.map(s => d3.format(".1%")(s.share))];
  }
  return htl.html`<div class="ex-table"><table class="table table-sm">
    <thead><tr>${splitActive ? htl.html`<th>Survey</th>` : null}<th>${byLabel}</th>${head.map(h => htl.html`<th class="num">${h}</th>`)}</tr></thead>
    <tbody>${rows.map(d => htl.html`<tr>${splitActive ? htl.html`<td>${d.survey}</td>` : null}<td>${d.group}</td>${cells(d).map(c => htl.html`<td class="num">${c}</td>`)}</tr>`)}</tbody>
  </table></div>`;
}

// Keep the controls and the address of the page in step --------------------

splitHint = {
  const el = viewof split;
  el.hint.hidden = !(split && surveyComparison);
  return surveyComparison;
}

dkVisible = {
  const el = viewof showDk;
  el.hidden = !dkLevel;
  return Boolean(dkLevel);
}

urlSync = {
  const p = new URLSearchParams();
  p.set("var", varName);
  p.set("by", by);
  if (!split) p.set("split", "0");
  if (surveys.length && surveys.length < datasets.length) p.set("surveys", surveys.map(datasetKey).join(","));
  if (ageRange[0] !== ageExtent[0] || ageRange[1] !== ageExtent[1]) p.set("age", ageRange.join("-"));
  if (once) p.set("once", "1");
  if (dkLevel && !showDk) p.set("dk", "0");
  history.replaceState(null, "", `${window.location.pathname}?${p}${window.location.hash}`);
  return p.toString();
}
